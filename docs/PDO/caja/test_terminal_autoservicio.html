127    def test_genera_codigo_de_8_digitos_pendiente(self):
128        """La operación queda pendiente, con código numérico y vencimiento futuro."""
129        op = self.generar()

138
139    def test_busqueda_exige_el_documento_correcto(self):
140        """El código solo sirve junto con el documento del cliente."""
141        op = self.generar()
142        self.assertEqual(tauser.buscar_operacion(op.codigo, '5550001').pk, op.pk)
143        with self.assertRaisesMessage(tauser.TerminalError, 'incorrectos'):
144            tauser.buscar_operacion(op.codigo, '9999999')
145        with self.assertRaisesMessage(tauser.TerminalError, 'incorrectos'):
146            tauser.buscar_operacion('00000000', '5550001')

148    def test_codigo_vencido(self):
149        """Un código vencido se marca VENCIDA y no se puede usar."""
150        op = self.generar()
151        OperacionTerminal.objects.filter(pk=op.pk).update(expira_en=timezone.now() - timedelta(minutes=1))
152        with self.assertRaisesMessage(tauser.TerminalError, 'venció'):
153            tauser.buscar_operacion(op.codigo, '5550001')
154        op.refresh_from_db()
155        self.assertEqual(op.estado, OperacionTerminal.VENCIDA)
156
157    def test_cancelacion(self):
158        """Una operación cancelada ya no puede completarse ni volver a cancelarse."""
159        op = self.generar()
160        tauser.cancelar_operacion(op, 'Prueba')
161        op.refresh_from_db()
162        self.assertEqual(op.estado, OperacionTerminal.CANCELADA)
163        with self.assertRaises(tauser.TerminalError):
164            tauser.cancelar_operacion(op, 'Otra vez')
165        with self.assertRaisesMessage(tauser.TerminalError, 'cancelada'):

168    def test_cambio_de_cotizacion(self):
169        """Si la tasa cambia, la terminal detecta la diferencia y el cliente puede aceptar la nueva cotización."""

178        op.refresh_from_db()
179        self.assertEqual(op.monto_destino, nueva.monto_destino)

186    def test_vender_dolares_con_monto_exacto(self):
187        """El cliente deposita 100 USD y recibe 726.000 Gs; el stock y la transacción se actualizan."""
188        op = self.generar('COMPRA', 'USD', '100')
189        op, tx = tauser.completar_operacion(op, self.terminal, {'100': 1})
190        self.assertEqual(op.estado, OperacionTerminal.COMPLETADA)
191        self.assertEqual(op.terminal, self.terminal)
192        self.assertEqual(op.vuelto, D('0'))
193        self.assertEqual(self.stock('USD', 100), 21)
194        # 726.000 = 7 x 100.000 + 20.000 + 5.000 + 1.000
195        self.assertEqual(op.billetes_entregados, {'PYG': {'100000': 7, '20000': 1, '5000': 1, '1000': 1}})
196        self.assertEqual(self.stock('PYG', 100000), 13)
197        self.assertIsInstance(tx, CurrencyPurchaseTransaction)
198        self.assertEqual(tx.payment_method_type, tauser.PAGO_TERMINAL)
199        self.assertEqual(tx.status, 'SUCCESS')
200        self.assertEqual(tx.converted_amount, D('726000.00'))
201        self.assertEqual(tx.total_pyg, D('726000.00'))
202        self.assertIn(op.codigo, tx.transparent_breakdown)
203
204    def test_comprar_dolares_con_vuelto(self):
205        """El cliente deposita de más en Guaraníes y recibe los dólares más el vuelto."""
206        op = self.generar('VENTA', 'USD', '50')
207        a_pagar = op.monto_origen
208        billetes_100k = int(a_pagar // 100000) + 1
209        op, tx = tauser.completar_operacion(op, self.terminal, {'100000': billetes_100k})
210        self.assertIsInstance(tx, CurrencySaleTransaction)
211        self.assertEqual(op.vuelto, D(billetes_100k * 100000) - a_pagar)
212        self.assertEqual(op.billetes_entregados['USD'], {'50': 1})
213        self.assertIn('vuelto_PYG', op.billetes_entregados)
214        self.assertEqual(self.stock('USD', 50), 19)
215
216    def test_deposito_insuficiente(self):
217        """Si no alcanza el depósito, la operación sigue pendiente y el stock no cambia."""
218        op = self.generar('COMPRA', 'USD', '100')
219        with self.assertRaisesMessage(tauser.TerminalError, 'debe depositar'):
220            tauser.completar_operacion(op, self.terminal, {'50': 1})
221        op.refresh_from_db()
222        self.assertEqual(op.estado, OperacionTerminal.PENDIENTE)
223        self.assertEqual(self.stock('USD', 50), 20)

318    def login_cliente(self):

325    def test_cliente_genera_y_cancela_codigo(self):

330        op = OperacionTerminal.objects.get()
331        self.assertContains(respuesta, op.codigo)
332        self.http.post(self.url_cliente, {'accion': 'cancelar', 'operacion_id': op.pk})
333        op.refresh_from_db()
334        self.assertEqual(op.estado, OperacionTerminal.CANCELADA)

336    def test_cliente_ve_error_de_monto(self):

340        self.assertContains(respuesta, 'billetes')

343    def test_pagina_del_cliente_requiere_inicio_de_sesion(self):
344        """Sin sesión se redirige al inicio de sesión."""
345        self.assertEqual(self.http.get(self.url_cliente).status_code, 302)

347    def test_kiosco_flujo_completo(self):
348        """En la terminal, sin iniciar sesión: código + documento, depósito y comprobante."""
349        op = self.generar('COMPRA', 'USD', '100')

354        self.assertContains(pantalla, 'Inserte sus billetes de USD')

357        self.assertContains(final, 'Operación completada')
358        op.refresh_from_db()
359        self.assertEqual(op.estado, OperacionTerminal.COMPLETADA)
