caja.terminal

Terminal de autoservicio (TAUser) para operaciones en efectivo ("metálico").

Flujo:

  1. El cliente genera la operación desde la web (generar_operacion): elige si vende o compra divisas, la divisa y el monto. Recibe un código de 8 dígitos válido por TERMINAL_VALIDEZ_MINUTOS.
  2. En la terminal ingresa el código y su documento (buscar_operacion). Si la cotización cambió desde que generó el código, debe aceptar la nueva (actualizar_cotizacion) o cancelar (cancelar_operacion).
  3. Deposita los billetes (completar_operacion): la terminal valida el monto, guarda los billetes en su stock, entrega el monto a recibir y el vuelto con los billetes disponibles y registra la transacción de compra o venta (EFECTIVO_TERMINAL), que entra en reportes, tablero de ganancias y facturación.

Los montos en efectivo se ajustan a billetes: el monto en Guaraníes se redondea al billete más chico (1.000 Gs) a favor de la casa, y el monto en divisa extranjera debe poder formarse con billetes.

 78def desglosar(monto, disponibles):
 79    """
 80    Arma un monto con billetes, respetando cuántos hay de cada valor. Usa primero los billetes más grandes
 81    y, si con eso no alcanza, busca otra combinación posible.
 82
 83    Args:
 84        monto (Decimal): Monto a formar.
 85        disponibles (dict[Decimal, int | None]): Billetes disponibles por valor facial
 86            (``None`` = cantidad ilimitada, para validar si un monto es formable).
 87
 88    Returns:
 89        dict[Decimal, int] | None: Cantidad de billetes por valor, o None si no se puede formar.
 90    """
 91    monto = Decimal(monto)
 92    if monto == 0:
 93        return {}
 94    if monto < 0 or monto != monto.to_integral_value():
 95        return None
 96    valores = sorted((Decimal(v) for v in disponibles if disponibles[v] != 0), reverse=True)
 97    if not valores or any(v != v.to_integral_value() for v in valores):
 98        return None
 99
100    # 1) Billetes más grandes primero (resuelve casi todos los casos).
101    resto, resultado = monto, {}
102    for valor in valores:
103        maximo = disponibles[valor]
104        cantidad = int(resto // valor)
105        if maximo is not None:
106            cantidad = min(cantidad, maximo)
107        if cantidad:
108            resultado[valor] = cantidad
109            resto -= valor * cantidad
110    if resto == 0:
111        return resultado
112
113    # 2) Búsqueda exacta (cambio con cantidades limitadas) sobre la unidad común de los billetes.
114    unidad = 0
115    for valor in valores:
116        unidad = gcd(unidad, int(valor))
117    if int(monto) % unidad:
118        return None
119    objetivo = int(monto) // unidad
120    if objetivo > MAXIMO_UNIDADES_DESGLOSE:
121        return None
122    pasos = [int(v) // unidad for v in valores]
123    alcanzado = [-1] * (objetivo + 1)  # índice del billete usado para llegar a cada monto
124    alcanzado[0] = len(valores)
125    for indice, paso in enumerate(pasos):
126        maximo = disponibles[valores[indice]]
127        usados = [0] * (objetivo + 1)
128        for x in range(paso, objetivo + 1):
129            if alcanzado[x] == -1 and alcanzado[x - paso] != -1 and (maximo is None or usados[x - paso] < maximo):
130                alcanzado[x] = indice
131                usados[x] = usados[x - paso] + 1
132    if alcanzado[objetivo] == -1:
133        return None
134    resultado, x = {}, objetivo
135    while x > 0:
136        indice = alcanzado[x]
137        resultado[valores[indice]] = resultado.get(valores[indice], 0) + 1
138        x -= pasos[indice]
139    return resultado

142def es_formable(monto, moneda):
143    """
144    Indica si un monto puede formarse con los billetes de la moneda (sin límite de cantidad).
145
146    Args:
147        monto (Decimal): Monto a validar.
148        moneda (str): Código de la divisa.
149
150    Returns:
151        bool: True si existe una combinación de billetes.
152    """
153    return desglosar(monto, {d.valor_facial: None for d in denominaciones(moneda)}) is not None

215    monto_origen: Decimal
216    monto_destino: Decimal
217    tasa: Decimal
218    tasa_std: Decimal
219    benefit_percentage: Decimal
220    comision: Decimal
221    valor_pyg: Decimal
222    redondeo_pyg: Decimal

225def cotizar(cliente, tipo_operacion, divisa, monto):
226    """
227    Cotiza una operación de terminal y ajusta los montos a billetes.
228
229    * ``COMPRA`` (el cliente vende divisas): ``monto`` es la divisa que deposita; recibe Guaraníes
230      redondeados hacia abajo al billete más chico.
231    * ``VENTA`` (el cliente compra divisas): ``monto`` es la divisa que recibe; deposita Guaraníes
232      redondeados hacia arriba al billete más chico.
233
234    Args:
235        cliente (Cliente): Cliente de la operación.
236        tipo_operacion (str): ``COMPRA`` o ``VENTA``.
237        divisa (str): Divisa extranjera (USD, EUR, BRL, ARS).
238        monto (Decimal): Monto en la divisa extranjera.
239
240    Returns:
241        CotizacionTerminal: Cotización ajustada.
242
243    Raises:
244        TerminalError: Si los datos no son válidos o el monto no se puede formar con billetes.
245    """
246    if tipo_operacion not in ('COMPRA', 'VENTA'):
247        raise TerminalError('Tipo de operación no válido.')
248    if not divisa or divisa == 'PYG':
249        raise TerminalError('Seleccione una divisa extranjera.')
250    try:
251        monto = Decimal(monto)
252    except Exception:
253        raise TerminalError('Ingrese un monto válido.')
254    if monto <= 0:
255        raise TerminalError('El monto debe ser mayor a cero.')
256    if not denominaciones(divisa):
257        raise TerminalError(f'La terminal no opera billetes de {divisa}.')
258    if not es_formable(monto, divisa):
259        minimo = billete_minimo(divisa)
260        raise TerminalError(
261            f'El monto de {divisa} debe poder pagarse con billetes (sin centavos; billete más chico: {minimo.normalize():f} {divisa}).'
262        )
263    billete_pyg = billete_minimo('PYG') or Decimal('1')
264
265    benefit_percentage, _ = beneficio_de_cliente(cliente)
266    if tipo_operacion == 'COMPRA':
267        moneda_origen, moneda_destino = divisa, 'PYG'
268    else:
269        moneda_origen, moneda_destino = 'PYG', divisa
270    base = cotizar_efectivo(tipo_operacion, moneda_origen, moneda_destino, monto, benefit_percentage)
271
272    if tipo_operacion == 'COMPRA':
273        exacto = base.monto_destino
274        ajustado = (exacto / billete_pyg).to_integral_value(rounding=ROUND_FLOOR) * billete_pyg
275        monto_origen, monto_destino, redondeo = monto, ajustado, exacto - ajustado
276        if monto_destino <= 0:
277            raise TerminalError('El monto es demasiado chico para entregarlo en billetes de Guaraníes.')
278    else:
279        exacto = base.monto_origen
280        ajustado = (exacto / billete_pyg).to_integral_value(rounding=ROUND_CEILING) * billete_pyg
281        monto_origen, monto_destino, redondeo = ajustado, monto, ajustado - exacto
282
283    return CotizacionTerminal(
284        tipo_operacion=tipo_operacion, moneda_origen=moneda_origen, moneda_destino=moneda_destino,
285        monto_origen=monto_origen.quantize(Decimal('0.01')), monto_destino=monto_destino.quantize(Decimal('0.01')),
286        tasa=base.tasa, tasa_std=base.tasa_std, benefit_percentage=benefit_percentage,
287        comision=base.comision.quantize(Decimal('0.01')), valor_pyg=base.valor_pyg.quantize(Decimal('0.01')),
288        redondeo_pyg=redondeo.quantize(Decimal('0.01')),
289    )

292def _aplicar_cotizacion(operacion, cotizacion):
293    """Copia los montos de una cotización en la operación (sin guardar)."""
294    for campo in ('tipo_operacion', 'moneda_origen', 'moneda_destino', 'monto_origen', 'monto_destino',
295                  'benefit_percentage', 'comision', 'valor_pyg', 'redondeo_pyg'):
296        setattr(operacion, campo, getattr(cotizacion, campo))
297    operacion.tasa_aplicada = cotizacion.tasa
298    operacion.tasa_estandar = cotizacion.tasa_std

305def _nuevo_codigo():
306    """Código numérico de 8 dígitos que no está en uso por otra operación pendiente."""
307    for _ in range(20):
308        codigo = f'{secrets.randbelow(10 ** 8):08d}'
309        if not OperacionTerminal.objects.filter(codigo=codigo, estado=OperacionTerminal.PENDIENTE).exists():
310            return codigo
311    raise TerminalError('No se pudo generar un código de operación. Intente nuevamente.')

314def generar_operacion(usuario, cliente, tipo_operacion, divisa, monto):
315    """
316    Crea una operación pendiente y su código para completarla en una terminal.
317
318    Args:
319        usuario (User): Usuario que la genera.
320        cliente (Cliente): Cliente activo.
321        tipo_operacion (str): ``COMPRA`` (vender divisas) o ``VENTA`` (comprar divisas).
322        divisa (str): Divisa extranjera.
323        monto (Decimal): Monto en la divisa extranjera.
324
325    Returns:
326        OperacionTerminal: Operación pendiente con su código.
327
328    Raises:
329        TerminalError: Si no hay cliente o la cotización no es válida.
330    """
331    if cliente is None:
332        raise TerminalError('Debe seleccionar un cliente activo para operar en la terminal.')
333    cotizacion = cotizar(cliente, tipo_operacion, divisa, monto)
334    operacion = OperacionTerminal(
335        codigo=_nuevo_codigo(), cliente=cliente, usuario=usuario,
336        expira_en=timezone.now() + timedelta(minutes=validez_minutos()),
337    )
338    _aplicar_cotizacion(operacion, cotizacion)
339    operacion.save()
340    return operacion

355def buscar_operacion(codigo, documento):
356    """
357    Busca la operación pendiente que el cliente quiere completar en la terminal.
358
359    Args:
360        codigo (str): Código de 8 dígitos.
361        documento (str): Cédula o RUC del cliente (segundo factor).
362
363    Returns:
364        OperacionTerminal: Operación pendiente y vigente.
365
366    Raises:
367        TerminalError: Si no existe, no coincide el documento, venció o ya no está pendiente.
368    """
369    codigo = (codigo or '').strip()
370    documento = (documento or '').strip()
371    vencer_operaciones()
372    operacion = (
373        OperacionTerminal.objects.filter(codigo=codigo, cliente__documento_identidad__iexact=documento)
374        .select_related('cliente').order_by('-creado_en').first()
375    )
376    if operacion is None or not codigo or not documento:
377        raise TerminalError('Código o documento incorrectos.')
378    if operacion.estado == OperacionTerminal.VENCIDA:
379        raise TerminalError('El código venció. Genere una nueva operación desde la web.')
380    if operacion.estado != OperacionTerminal.PENDIENTE:
381        raise TerminalError(f'La operación ya está {operacion.get_estado_display().lower()}.')
382    return operacion

385def cotizacion_cambio(operacion):
386    """
387    Recalcula la operación con la cotización vigente.
388
389    Args:
390        operacion (OperacionTerminal): Operación pendiente.
391
392    Returns:
393        CotizacionTerminal | None: Nueva cotización si los montos cambiaron, o None si sigue igual.
394    """
395    monto = operacion.monto_origen if operacion.tipo_operacion == 'COMPRA' else operacion.monto_destino
396    nueva = cotizar(operacion.cliente, operacion.tipo_operacion, operacion.divisa_extranjera, monto)
397    if nueva.monto_origen == operacion.monto_origen and nueva.monto_destino == operacion.monto_destino:
398        return None
399    return nueva

419def cancelar_operacion(operacion, motivo):
420    """
421    Cancela una operación pendiente.
422
423    Args:
424        operacion (OperacionTerminal): Operación a cancelar.
425        motivo (str): Motivo de la cancelación.
426
427    Raises:
428        TerminalError: Si la operación ya no está pendiente.
429    """
430    if operacion.estado != OperacionTerminal.PENDIENTE:

433    operacion.motivo_cancelacion = (motivo or '')[:200]
434    operacion.save(update_fields=['estado', 'motivo_cancelacion'])

437def _leer_billetes(moneda, billetes):
438    """
439    Normaliza los billetes depositados a ``{Decimal: cantidad}`` y valida que existan en la moneda.
440
441    Raises:
442        TerminalError: Si hay cantidades negativas o denominaciones que no son de la moneda.
443    """
444    validos = {d.valor_facial for d in denominaciones(moneda)}
445    resultado = {}
446    for valor, cantidad in (billetes or {}).items():
447        try:
448            valor, cantidad = Decimal(str(valor)), int(cantidad or 0)
449        except Exception:
450            raise TerminalError('Cantidad de billetes no válida.')
451        if cantidad < 0:
452            raise TerminalError('La cantidad de billetes no puede ser negativa.')
453        if cantidad == 0:
454            continue
455        if valor not in validos:
456            raise TerminalError(f'La terminal no acepta billetes de {valor.normalize():f} {moneda}.')
457        resultado[valor] = resultado.get(valor, 0) + cantidad
458    return resultado

469def _registrar_transaccion(operacion):
470    """
471    Registra la compra o venta de la operación, igual que la ventanilla pero con ``EFECTIVO_TERMINAL``.
472
473    Returns:
474        CurrencyPurchaseTransaction | CurrencySaleTransaction: Transacción creada.
475    """
476    datos = dict(
477        user=operacion.usuario, cliente=operacion.cliente, turno=None,
478        from_currency=operacion.moneda_origen, to_currency=operacion.moneda_destino,
479        amount=operacion.monto_origen, converted_amount=operacion.monto_destino,
480        applied_rate=operacion.tasa_aplicada, standard_rate=operacion.tasa_estandar,
481        benefit_percentage=operacion.benefit_percentage, commission_amount=operacion.comision,
482        status='SUCCESS', payment_method_type=PAGO_TERMINAL, monto_moneda_local=operacion.valor_pyg,
483        transparent_breakdown=(
484            f"{MARCA_TERMINAL} {operacion.terminal.codigo} Operación: {operacion.codigo} | "
485            f"Redondeo a billetes: {operacion.redondeo_pyg} PYG"
486        ),
487    )
488    if operacion.tipo_operacion == 'COMPRA':
489        return CurrencyPurchaseTransaction.objects.create(total_pyg=operacion.monto_destino, **datos)
490    return CurrencySaleTransaction.objects.create(total_pyg=operacion.monto_origen, **datos)

493def completar_operacion(operacion, terminal, billetes):
494    """
495    Completa la operación en la terminal: recibe los billetes, entrega el monto y el vuelto,
496    actualiza el stock y registra la transacción. Todo o nada: si la terminal no puede entregar
497    el dinero, no se modifica nada y el cliente retira sus billetes.
498
499    Args:
500        operacion (OperacionTerminal): Operación pendiente.
501        terminal (TerminalAutoservicio): Terminal donde se completa.
502        billetes (dict): Billetes depositados ``{valor_facial: cantidad}`` en la moneda de origen.
503
504    Returns:
505        tuple[OperacionTerminal, CurrencyPurchaseTransaction | CurrencySaleTransaction]: Operación completada y transacción.
506
507    Raises:
508        TerminalError: Si la terminal está inactiva, el depósito no alcanza o no hay billetes para entregar.
509    """
510    if not terminal.activa:
511        raise TerminalError('La terminal está fuera de servicio.')
512    depositados = _leer_billetes(operacion.moneda_origen, billetes)
513    total_depositado = sum((valor * cantidad for valor, cantidad in depositados.items()), Decimal('0'))
514    if total_depositado < operacion.monto_origen:
515        raise TerminalError(
516            f'Depositó {total_depositado:,.2f} {operacion.moneda_origen}; '
517            f'debe depositar {operacion.monto_origen:,.2f} {operacion.moneda_origen}.'
518        )
519    vuelto = total_depositado - operacion.monto_origen
520
521    if OperacionTerminal.objects.filter(
522        pk=operacion.pk, estado=OperacionTerminal.PENDIENTE, expira_en__lt=timezone.now()
523    ).update(estado=OperacionTerminal.VENCIDA):
524        raise TerminalError('El código venció. Genere una nueva operación desde la web.')
525
526    with transaction.atomic():
527        operacion = OperacionTerminal.objects.select_for_update().get(pk=operacion.pk)
528        if operacion.estado != OperacionTerminal.PENDIENTE:
529            raise TerminalError(f'La operación ya está {operacion.get_estado_display().lower()}.')
530
531        asegurar_stock(terminal)
532        stock_origen = stock_de(terminal, operacion.moneda_origen, bloquear=True)
533        stock_destino = stock_de(terminal, operacion.moneda_destino, bloquear=True)
534
535        # Los billetes depositados entran al stock y pueden usarse para el vuelto.
536        for valor, cantidad in depositados.items():
537            stock_origen[valor].cantidad += cantidad
538            stock_origen[valor].save(update_fields=['cantidad'])
539
540        entrega = desglosar(operacion.monto_destino, {v: f.cantidad for v, f in stock_destino.items()})
541        if entrega is None:  # la excepción revierte el depósito: el cliente retira sus billetes
542            raise TerminalError(
543                f'La terminal no tiene billetes de {operacion.moneda_destino} suficientes para entregar '
544                f'{operacion.monto_destino:,.2f}. Retire sus billetes y diríjase a ventanilla.'
545            )
546        _retirar(stock_destino, entrega)
547
548        entrega_vuelto = desglosar(vuelto, {v: f.cantidad for v, f in stock_origen.items()})
549        if entrega_vuelto is None:
550            raise TerminalError(
551                f'La terminal no tiene billetes para entregar el vuelto de {vuelto:,.2f} {operacion.moneda_origen}. '
552                'Retire sus billetes e intente con el monto exacto.'
553            )
554        _retirar(stock_origen, entrega_vuelto)
555
556        operacion.terminal = terminal
557        operacion.estado = OperacionTerminal.COMPLETADA
558        operacion.efectivo_depositado = total_depositado
559        operacion.vuelto = vuelto
560        operacion.billetes_depositados = _a_json(depositados)
561        operacion.billetes_entregados = {operacion.moneda_destino: _a_json(entrega)}
562        if entrega_vuelto:
563            operacion.billetes_entregados[f'vuelto_{operacion.moneda_origen}'] = _a_json(entrega_vuelto)
564        operacion.completado_en = timezone.now()
565        operacion.save()
566
567        transaccion = _registrar_transaccion(operacion)
568        cliente = operacion.cliente
569        cliente.transaction_volume = Decimal(str(cliente.transaction_volume or 0)) + (
570            operacion.monto_destino if operacion.tipo_operacion == 'COMPRA' else operacion.monto_origen
571        )
572        cliente.save(update_fields=['transaction_volume'])
573    return operacion, transaccion

def desglosar(monto, disponibles): 
 79def desglosar(monto, disponibles):
 80    """
 81    Arma un monto con billetes, respetando cuántos hay de cada valor. Usa primero los billetes más grandes
 82    y, si con eso no alcanza, busca otra combinación posible.
 83
 84    Args:
 85        monto (Decimal): Monto a formar.
 86        disponibles (dict[Decimal, int | None]): Billetes disponibles por valor facial
 87            (``None`` = cantidad ilimitada, para validar si un monto es formable).
 88
 89    Returns:
 90        dict[Decimal, int] | None: Cantidad de billetes por valor, o None si no se puede formar.
 91    """
 92    monto = Decimal(monto)
 93    if monto == 0:
 94        return {}
 95    if monto < 0 or monto != monto.to_integral_value():
 96        return None
 97    valores = sorted((Decimal(v) for v in disponibles if disponibles[v] != 0), reverse=True)
 98    if not valores or any(v != v.to_integral_value() for v in valores):
 99        return None
100
101    # 1) Billetes más grandes primero (resuelve casi todos los casos).
102    resto, resultado = monto, {}
103    for valor in valores:
104        maximo = disponibles[valor]
105        cantidad = int(resto // valor)
106        if maximo is not None:
107            cantidad = min(cantidad, maximo)
108        if cantidad:
109            resultado[valor] = cantidad
110            resto -= valor * cantidad
111    if resto == 0:
112        return resultado
113
114    # 2) Búsqueda exacta (cambio con cantidades limitadas) sobre la unidad común de los billetes.
115    unidad = 0
116    for valor in valores:
117        unidad = gcd(unidad, int(valor))
118    if int(monto) % unidad:
119        return None
120    objetivo = int(monto) // unidad
121    if objetivo > MAXIMO_UNIDADES_DESGLOSE:
122        return None
123    pasos = [int(v) // unidad for v in valores]
124    alcanzado = [-1] * (objetivo + 1)  # índice del billete usado para llegar a cada monto
125    alcanzado[0] = len(valores)
126    for indice, paso in enumerate(pasos):
127        maximo = disponibles[valores[indice]]
128        usados = [0] * (objetivo + 1)
129        for x in range(paso, objetivo + 1):
130            if alcanzado[x] == -1 and alcanzado[x - paso] != -1 and (maximo is None or usados[x - paso] < maximo):
131                alcanzado[x] = indice
132                usados[x] = usados[x - paso] + 1
133    if alcanzado[objetivo] == -1:
134        return None
135    resultado, x = {}, objetivo
136    while x > 0:
137        indice = alcanzado[x]
138        resultado[valores[indice]] = resultado.get(valores[indice], 0) + 1
139        x -= pasos[indice]
140    return resultado

226def cotizar(cliente, tipo_operacion, divisa, monto):
227    """
228    Cotiza una operación de terminal y ajusta los montos a billetes.
229
230    * ``COMPRA`` (el cliente vende divisas): ``monto`` es la divisa que deposita; recibe Guaraníes
231      redondeados hacia abajo al billete más chico.
232    * ``VENTA`` (el cliente compra divisas): ``monto`` es la divisa que recibe; deposita Guaraníes
233      redondeados hacia arriba al billete más chico.
234
235    Args:
236        cliente (Cliente): Cliente de la operación.
237        tipo_operacion (str): ``COMPRA`` o ``VENTA``.
238        divisa (str): Divisa extranjera (USD, EUR, BRL, ARS).
239        monto (Decimal): Monto en la divisa extranjera.
240
241    Returns:
242        CotizacionTerminal: Cotización ajustada.
243
244    Raises:
245        TerminalError: Si los datos no son válidos o el monto no se puede formar con billetes.
246    """
247    if tipo_operacion not in ('COMPRA', 'VENTA'):
248        raise TerminalError('Tipo de operación no válido.')
249    if not divisa or divisa == 'PYG':
250        raise TerminalError('Seleccione una divisa extranjera.')
251    try:
252        monto = Decimal(monto)
253    except Exception:
254        raise TerminalError('Ingrese un monto válido.')
255    if monto <= 0:
256        raise TerminalError('El monto debe ser mayor a cero.')
257    if not denominaciones(divisa):
258        raise TerminalError(f'La terminal no opera billetes de {divisa}.')
259    if not es_formable(monto, divisa):
260        minimo = billete_minimo(divisa)
261        raise TerminalError(
262            f'El monto de {divisa} debe poder pagarse con billetes (sin centavos; billete más chico: {minimo.normalize():f} {divisa}).'
263        )
264    billete_pyg = billete_minimo('PYG') or Decimal('1')
265
266    benefit_percentage, _ = beneficio_de_cliente(cliente)
267    if tipo_operacion == 'COMPRA':
268        moneda_origen, moneda_destino = divisa, 'PYG'
269    else:
270        moneda_origen, moneda_destino = 'PYG', divisa
271    base = cotizar_efectivo(tipo_operacion, moneda_origen, moneda_destino, monto, benefit_percentage)
272
273    if tipo_operacion == 'COMPRA':
274        exacto = base.monto_destino
275        ajustado = (exacto / billete_pyg).to_integral_value(rounding=ROUND_FLOOR) * billete_pyg
276        monto_origen, monto_destino, redondeo = monto, ajustado, exacto - ajustado
277        if monto_destino <= 0:
278            raise TerminalError('El monto es demasiado chico para entregarlo en billetes de Guaraníes.')
279    else:
280        exacto = base.monto_origen
281        ajustado = (exacto / billete_pyg).to_integral_value(rounding=ROUND_CEILING) * billete_pyg
282        monto_origen, monto_destino, redondeo = ajustado, monto, ajustado - exacto
283
284    return CotizacionTerminal(
285        tipo_operacion=tipo_operacion, moneda_origen=moneda_origen, moneda_destino=moneda_destino,
286        monto_origen=monto_origen.quantize(Decimal('0.01')), monto_destino=monto_destino.quantize(Decimal('0.01')),
287        tasa=base.tasa, tasa_std=base.tasa_std, benefit_percentage=benefit_percentage,
288        comision=base.comision.quantize(Decimal('0.01')), valor_pyg=base.valor_pyg.quantize(Decimal('0.01')),
289        redondeo_pyg=redondeo.quantize(Decimal('0.01')),
290    )

def completar_operacion(operacion, terminal, billetes): 
494def completar_operacion(operacion, terminal, billetes):
495    """
496    Completa la operación en la terminal: recibe los billetes, entrega el monto y el vuelto,
497    actualiza el stock y registra la transacción. Todo o nada: si la terminal no puede entregar
498    el dinero, no se modifica nada y el cliente retira sus billetes.
499
500    Args:
501        operacion (OperacionTerminal): Operación pendiente.
502        terminal (TerminalAutoservicio): Terminal donde se completa.
503        billetes (dict): Billetes depositados ``{valor_facial: cantidad}`` en la moneda de origen.
504
505    Returns:
506        tuple[OperacionTerminal, CurrencyPurchaseTransaction | CurrencySaleTransaction]: Operación completada y transacción.
507
508    Raises:
509        TerminalError: Si la terminal está inactiva, el depósito no alcanza o no hay billetes para entregar.
510    """
511    if not terminal.activa:
512        raise TerminalError('La terminal está fuera de servicio.')
513    depositados = _leer_billetes(operacion.moneda_origen, billetes)
514    total_depositado = sum((valor * cantidad for valor, cantidad in depositados.items()), Decimal('0'))
515    if total_depositado < operacion.monto_origen:
516        raise TerminalError(
517            f'Depositó {total_depositado:,.2f} {operacion.moneda_origen}; '
518            f'debe depositar {operacion.monto_origen:,.2f} {operacion.moneda_origen}.'
519        )
520    vuelto = total_depositado - operacion.monto_origen
521
522    if OperacionTerminal.objects.filter(
523        pk=operacion.pk, estado=OperacionTerminal.PENDIENTE, expira_en__lt=timezone.now()
524    ).update(estado=OperacionTerminal.VENCIDA):
525        raise TerminalError('El código venció. Genere una nueva operación desde la web.')
526
527    with transaction.atomic():
528        operacion = OperacionTerminal.objects.select_for_update().get(pk=operacion.pk)
529        if operacion.estado != OperacionTerminal.PENDIENTE:
530            raise TerminalError(f'La operación ya está {operacion.get_estado_display().lower()}.')
531
532        asegurar_stock(terminal)
533        stock_origen = stock_de(terminal, operacion.moneda_origen, bloquear=True)
534        stock_destino = stock_de(terminal, operacion.moneda_destino, bloquear=True)
535
536        # Los billetes depositados entran al stock y pueden usarse para el vuelto.
537        for valor, cantidad in depositados.items():
538            stock_origen[valor].cantidad += cantidad
539            stock_origen[valor].save(update_fields=['cantidad'])
540
541        entrega = desglosar(operacion.monto_destino, {v: f.cantidad for v, f in stock_destino.items()})
542        if entrega is None:  # la excepción revierte el depósito: el cliente retira sus billetes
543            raise TerminalError(
544                f'La terminal no tiene billetes de {operacion.moneda_destino} suficientes para entregar '
545                f'{operacion.monto_destino:,.2f}. Retire sus billetes y diríjase a ventanilla.'
546            )
547        _retirar(stock_destino, entrega)
548
549        entrega_vuelto = desglosar(vuelto, {v: f.cantidad for v, f in stock_origen.items()})
550        if entrega_vuelto is None:
551            raise TerminalError(
552                f'La terminal no tiene billetes para entregar el vuelto de {vuelto:,.2f} {operacion.moneda_origen}. '
553                'Retire sus billetes e intente con el monto exacto.'
554            )
555        _retirar(stock_origen, entrega_vuelto)
556
557        operacion.terminal = terminal
558        operacion.estado = OperacionTerminal.COMPLETADA
559        operacion.efectivo_depositado = total_depositado
560        operacion.vuelto = vuelto
561        operacion.billetes_depositados = _a_json(depositados)
562        operacion.billetes_entregados = {operacion.moneda_destino: _a_json(entrega)}
563        if entrega_vuelto:
564            operacion.billetes_entregados[f'vuelto_{operacion.moneda_origen}'] = _a_json(entrega_vuelto)
565        operacion.completado_en = timezone.now()
566        operacion.save()
567
568        transaccion = _registrar_transaccion(operacion)
569        cliente = operacion.cliente
570        cliente.transaction_volume = Decimal(str(cliente.transaction_volume or 0)) + (
571            operacion.monto_destino if operacion.tipo_operacion == 'COMPRA' else operacion.monto_origen
572        )
573        cliente.save(update_fields=['transaction_volume'])
574    return operacion, transaccion

577def recargar_stock(terminal, moneda, billetes, retirar=False):
578    """
579    Carga (o retira) billetes del stock de una terminal. Lo usa el administrador.
580
581    Args:
582        terminal (TerminalAutoservicio): Terminal.
583        moneda (str): Código de la divisa.
584        billetes (dict): ``{valor_facial: cantidad}``.

586

588        Decimal: Monto total cargado o retirado.
589

602                )

604            fila.save(update_fields=['cantidad'])
605    return sum((valor * cantidad for valor, cantidad in cantidades.items()), Decimal('0'))
