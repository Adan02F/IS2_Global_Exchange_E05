authentication.mfa

Verificación en dos pasos (iToken) con código enviado por correo electrónico.

En cada inicio de sesión de un usuario que requiere MFA (administrativos y corporativos) se genera un código aleatorio de 6 dígitos, se envía a su correo y se guarda en la sesión solo como hash HMAC. El código:

  • vence a los CODIGO_VIGENCIA_SEGUNDOS,
  • se puede usar una sola vez,
  • se invalida tras MAX_INTENTOS intentos fallidos,
  • se puede reenviar cada REENVIO_ESPERA_SEGUNDOS.

Mientras la verificación está pendiente, MFAPendienteMiddleware solo permite la pantalla del iToken y el cierre de sesión.

 46def generar_codigo():
 47    """Genera un código numérico aleatorio criptográficamente seguro de 6 dígitos."""
 48    return f'{secrets.randbelow(10 ** CODIGO_DIGITOS):0{CODIGO_DIGITOS}d}'

 51def _hash_codigo(codigo, user_id):
 52    """Hash HMAC-SHA256 del código ligado al usuario (el código nunca se guarda en claro)."""
 53    mensaje = f'{user_id}:{codigo}'.encode()
 54    return hmac.new(settings.SECRET_KEY.encode(), mensaje, hashlib.sha256).hexdigest()

 57def enmascarar_email(email):
 58    """Oculta parte del correo para mostrarlo en pantalla: diego@gmail.com -> d***o@gmail.com."""
 59    if not email or '@' not in email:
 60        return ''
 61    usuario, dominio = email.split('@', 1)
 62    if len(usuario) <= 2:
 63        return f'{usuario[0]}***@{dominio}'
 64    return f'{usuario[0]}***{usuario[-1]}@{dominio}'

 69    return bool(request.session.get(SESION_PENDIENTE))

 72def segundos_para_reenviar(request, ahora=None):
 73    """Segundos que faltan para poder pedir otro código (0 si ya se puede)."""
 74    ahora = time.time() if ahora is None else ahora
 75    enviado = request.session.get(SESION_ENVIADO)
 76    if not enviado:
 77        return 0
 78    return max(0, int(enviado + REENVIO_ESPERA_SEGUNDOS - ahora))

 81def iniciar_verificacion(request, user):
 82    """
 83    Marca la sesión como pendiente de verificación y envía un código nuevo por correo.
 84
 85    Returns:
 86        bool: True si el correo se envió; False si el usuario no tiene correo o falló el envío.
 87    """
 88    request.session[SESION_PENDIENTE] = True
 89    return enviar_codigo(request, user)

 92def enviar_codigo(request, user, ahora=None):
 93    """
 94    Genera un código nuevo (invalida el anterior), lo guarda en la sesión y lo envía por correo.
 95
 96    Returns:
 97        bool: True si el correo se envió correctamente.
 98    """
 99    ahora = time.time() if ahora is None else ahora
100    codigo = generar_codigo()
101    request.session[SESION_HASH] = _hash_codigo(codigo, user.pk)
102    request.session[SESION_EXPIRA] = ahora + CODIGO_VIGENCIA_SEGUNDOS
103    request.session[SESION_INTENTOS] = 0
104    request.session[SESION_ENVIADO] = ahora
105
106    if not user.email:
107        logger.warning('Usuario %s sin correo: no se puede enviar el código de verificación.', user.username)
108        return False
109    minutos = CODIGO_VIGENCIA_SEGUNDOS // 60
110    try:
111        send_mail(
112            subject='Global Exchange - Código de verificación',
113            message=(
114                f'Hola {user.get_full_name() or user.username},\n\n'
115                f'Tu código de verificación (iToken) es: {codigo}\n\n'
116                f'Vence en {minutos} minutos y solo puede usarse una vez.\n'
117                'Si no intentaste iniciar sesión, ignora este correo y avisa al administrador.\n\n'
118                'Global Exchange'
119            ),
120            from_email=settings.DEFAULT_FROM_EMAIL,
121            recipient_list=[user.email],
122        )
123    except Exception:  # noqa: BLE001 - cualquier fallo SMTP se informa al usuario sin romper el login
124        logger.exception('No se pudo enviar el código de verificación a %s.', user.username)
125        return False
126    return True

129def verificar_codigo(request, user, codigo, ahora=None):
130    """
131    Valida el código ingresado contra el guardado en la sesión.
132
133    Returns:
134        str: VALIDO, INVALIDO, EXPIRADO, BLOQUEADO o SIN_CODIGO.
135    """
136    ahora = time.time() if ahora is None else ahora
137    guardado = request.session.get(SESION_HASH)
138    if not guardado:
139        return SIN_CODIGO
140    if ahora > request.session.get(SESION_EXPIRA, 0):
141        _descartar_codigo(request)
142        return EXPIRADO
143    if request.session.get(SESION_INTENTOS, 0) >= MAX_INTENTOS:
144        _descartar_codigo(request)
145        return BLOQUEADO
146
147    codigo = (codigo or '').strip()
148    if codigo.isdigit() and hmac.compare_digest(guardado, _hash_codigo(codigo, user.pk)):
149        completar_verificacion(request)
150        return VALIDO
151
152    request.session[SESION_INTENTOS] = request.session.get(SESION_INTENTOS, 0) + 1
153    if request.session[SESION_INTENTOS] >= MAX_INTENTOS:
154        _descartar_codigo(request)
155        return BLOQUEADO
156    return INVALIDO
157

159def intentos_restantes(request):
160    """Cantidad de intentos que quedan para el código actual."""
161    return max(0, MAX_INTENTOS - request.session.get(SESION_INTENTOS, 0))

164def completar_verificacion(request):
165    """Quita la marca de pendiente y descarta el código (uso único)."""
166    request.session.pop(SESION_PENDIENTE, None)
167    _descartar_codigo(request)
168
169
170def _descartar_codigo(request):
171    """Elimina el código de la sesión (vencido, bloqueado o ya usado)."""
172    for clave in (SESION_HASH, SESION_EXPIRA, SESION_INTENTOS):
173        request.session.pop(clave, None)
