 32class SeedDataTests(TestCase):
 33    """Pruebas del comando de población inicial seed_data."""
 34
 35    def test_carga_todas_las_divisas(self):
 36        """Debe crear las 5 divisas base con sus tasas y símbolos."""
 37        seed_all()
 38        codigos = set(ExchangeRate.objects.values_list('currency_code', flat=True))
 39        self.assertEqual(codigos, {d[0] for d in DIVISAS_DEFAULT})
 40        usd = ExchangeRate.objects.get(currency_code='USD')
 41        self.assertEqual(usd.buy_rate, Decimal('7300.0000'))
 42        self.assertEqual(usd.sell_rate, Decimal('7450.0000'))
 43
 44    def test_carga_metodos_de_pago(self):
 45        """Debe crear los métodos de pago predeterminados (PSE-25)."""
 46        seed_all()
 47        codigos = set(PaymentMethod.objects.values_list('code', flat=True))
 48        self.assertTrue({'TARJETA_CREDITO', 'TARJETA_DEBITO', 'TRANSFERENCIA', 'BILLETERA', 'EFECTIVO'} <= codigos)
 49
 50    def test_carga_tabla_de_comisiones(self):
 51        """Debe crear las reglas de comisión/beneficio por categoría (PSE-29)."""
 52        seed_all()
 53        reglas = dict(ClientBenefitRule.objects.values_list('category_code', 'benefit_percentage'))
 54        self.assertEqual(reglas['MINORISTA'], Decimal('0.00'))
 55        self.assertEqual(reglas['VIP'], Decimal('2.00'))
 56        self.assertEqual(reglas['CORPORATIVO'], Decimal('4.00'))
 57
 58    def test_carga_denominaciones_y_cajas(self):
 59        """Debe crear las denominaciones de billetes y las cajas físicas C01/C02."""
 60        seed_all()
 61        self.assertTrue(DenominacionDivisa.objects.filter(divisa='PYG', valor_facial=Decimal('100000')).exists())
 62        self.assertEqual(set(Caja.objects.values_list('codigo', flat=True)), {'C01', 'C02'})
 63
 64    def test_crea_usuarios_de_prueba_con_rol(self):
 65        """Cada usuario de prueba debe existir con contraseña y el rol esperado."""
 66        seed_all()
 67        for data in USUARIOS_PRUEBA:
 68            user = User.objects.get(username=data['username'])
 69            self.assertTrue(user.check_password(PASSWORD_PRUEBA))
 70            self.assertEqual(UserProfile.objects.get(user=user).role.name, data['role'])
 71        self.assertTrue(Caja.objects.get(codigo='C01').cajeros_asignados.filter(username='cajero').exists())
 72        self.assertTrue(Cliente.objects.filter(documento_identidad='1234567').exists())
 73
 74    def test_es_idempotente(self):
 75        """Ejecutar dos veces no debe duplicar registros."""
 76        primera = seed_all()
 77        segunda = seed_all()
 78        self.assertEqual(primera, segunda)
 79
 80    def test_no_pisa_cambios_del_administrador(self):
 81        """Si el admin modificó una tasa o una contraseña, el seed no debe sobrescribirlas."""
 82        seed_all()
 83        ExchangeRate.objects.filter(currency_code='USD').update(buy_rate=Decimal('7000.0000'))
 84        user = User.objects.get(username='cajero')
 85        user.set_password('otra-clave')
 86        user.save()
 87
 88        seed_all()
 89
 90        self.assertEqual(ExchangeRate.objects.get(currency_code='USD').buy_rate, Decimal('7000.0000'))
 91        self.assertTrue(User.objects.get(username='cajero').check_password('otra-clave'))
 92
 93    def test_no_reemplaza_rol_existente(self):
 94        """Un usuario que ya tiene rol asignado conserva su rol."""
 95        otro_rol = Role.objects.create(name='Supervisor')
 96        user = User.objects.create_user(username='cajero', password='x')
 97        UserProfile.objects.create(user=user, role=otro_rol)
 98        seed_all()
 99        self.assertEqual(UserProfile.objects.get(user=user).role.name, 'Supervisor')
100
101    def test_sin_usuarios_solo_carga_catalogos(self):
102        """Con --sin-usuarios no se crean usuarios ni cliente demo, pero sí los catálogos."""
103        out = StringIO()
104        call_command('seed_data', '--sin-usuarios', stdout=out)
105        self.assertFalse(User.objects.exists())
106        self.assertFalse(Cliente.objects.exists())
107        self.assertEqual(ExchangeRate.objects.count(), len(DIVISAS_DEFAULT))
108        self.assertIn('Población inicial completada', out.getvalue())
109
110    def test_comando_imprime_resumen(self):
111        """El comando debe informar los usuarios de prueba disponibles."""
112        out = StringIO()
113        call_command('seed_data', stdout=out)
114        self.assertIn('adminuser', out.getvalue())

117def _resp(status=200, payload=None):
118    """Construye una respuesta HTTP simulada de requests."""
119    response = mock.Mock(status_code=status)
120    response.json.return_value = payload if payload is not None else {}
121    response.raise_for_status.return_value = None
122    return response

125class KeycloakAdminTests(SimpleTestCase):
126    """Pruebas del cliente de Admin API usado por keycloak_bootstrap."""
127
128    def setUp(self):
129        """Crea un cliente KeycloakAdmin ya autenticado."""
130        self.kc = KeycloakAdmin('http://keycloak:8080/', 'realm', 'client', 'secret')
131        self.kc.headers = {'Authorization': 'Bearer t'}
132
133    @mock.patch.object(keycloak_bootstrap.requests, 'post')
134    def test_authenticate_guarda_token(self, post):
135        """authenticate() debe usar client_credentials y guardar el bearer token."""
136        post.return_value = _resp(payload={'access_token': 'abc'})
137        self.kc.authenticate()
138        self.assertEqual(self.kc.headers['Authorization'], 'Bearer abc')
139        self.assertEqual(post.call_args.kwargs['data']['grant_type'], 'client_credentials')
140        self.assertTrue(post.call_args.args[0].startswith('http://keycloak:8080/realms/realm/'))
141
142    @mock.patch.object(keycloak_bootstrap.requests, 'put')
143    @mock.patch.object(keycloak_bootstrap.requests, 'get')
144    def test_agrega_urls_de_produccion_al_cliente(self, get, put):
145        """Debe agregar callback, origen y logout del dominio público si faltan."""
146        get.return_value = _resp(payload=[{
147            'id': 'uuid-1',
148            'redirectUris': ['http://localhost:8000/auth/callback/'],
149            'webOrigins': ['http://localhost:8000'],
150            'attributes': {'post.logout.redirect.uris': 'http://localhost:8000/auth/login/'},
151        }])
152        put.return_value = _resp()
153
154        changed = self.kc.ensure_client_urls('https://ge.duckdns.org/auth/callback/')
155
156        self.assertTrue(changed)
157        enviado = put.call_args.kwargs['json']
158        self.assertIn('https://ge.duckdns.org/auth/callback/', enviado['redirectUris'])
159        self.assertIn('http://localhost:8000/auth/callback/', enviado['redirectUris'])
160        self.assertIn('https://ge.duckdns.org', enviado['webOrigins'])
161        self.assertEqual(
162            enviado['attributes']['post.logout.redirect.uris'],
163            'http://localhost:8000/auth/login/##https://ge.duckdns.org/auth/login/',
164        )
165
166    @mock.patch.object(keycloak_bootstrap.requests, 'put')
167    @mock.patch.object(keycloak_bootstrap.requests, 'get')
168    def test_no_modifica_cliente_si_ya_esta_configurado(self, get, put):
169        """Si las URLs ya están registradas no debe hacer PUT."""
170        get.return_value = _resp(payload=[{
171            'id': 'uuid-1',
172            'redirectUris': ['http://localhost:8000/auth/callback/'],
173            'webOrigins': ['http://localhost:8000'],
174            'attributes': {'post.logout.redirect.uris': 'http://localhost:8000/auth/login/'},
175        }])
176        self.assertFalse(self.kc.ensure_client_urls('http://localhost:8000/auth/callback/'))
177        put.assert_not_called()
178
179    @mock.patch.object(keycloak_bootstrap.requests, 'get')
180    def test_cliente_inexistente_lanza_error(self, get):
181        """Si el cliente OIDC no existe en el realm debe lanzar RuntimeError."""
182        get.return_value = _resp(payload=[])
183        with self.assertRaises(RuntimeError):
184            self.kc.ensure_client_urls('https://ge.duckdns.org/auth/callback/')
185
186    @mock.patch.object(keycloak_bootstrap.requests, 'post')
187    @mock.patch.object(keycloak_bootstrap.requests, 'get')
188    def test_usuario_existente_no_se_recrea(self, get, post):
189        """Un usuario que ya existe en Keycloak no se vuelve a crear ni cambia su contraseña."""
190        get.return_value = _resp(payload=[{'id': 'u1', 'username': 'cajero'}])
191        self.assertFalse(self.kc.ensure_user('cajero', 'c@x.com', 'pw', 'Cajero'))
192        post.assert_not_called()
193
194    @mock.patch.object(keycloak_bootstrap.requests, 'post')
195    @mock.patch.object(keycloak_bootstrap.requests, 'get')
196    def test_crea_usuario_y_asigna_rol(self, get, post):
197        """Debe crear el usuario con contraseña no temporal y asignarle el rol de realm."""
198        rol = {'id': 'r1', 'name': 'Cajero'}
199        get.side_effect = [_resp(payload=[]), _resp(payload=[{'id': 'u1'}]), _resp(payload=rol)]
200        post.return_value = _resp(status=201)
201
202        self.assertTrue(self.kc.ensure_user('cajero', 'c@x.com', 'pw', 'Cajero'))
203
204        crear, asignar = post.call_args_list
205        credencial = crear.kwargs['json']['credentials'][0]
206        self.assertEqual(credencial, {'type': 'password', 'value': 'pw', 'temporary': False})
207        self.assertTrue(asignar.args[0].endswith('/users/u1/role-mappings/realm'))
208        self.assertEqual(asignar.kwargs['json'], [rol])

248class SettingsEnvTests(SimpleTestCase):
249    """Pruebas de los helpers que leen la configuración desde variables de entorno."""
250
251    def test_env_bool(self):
252        """Interpreta true/1/yes/on como verdadero y usa el default si falta."""
253        with mock.patch.dict(os.environ, {'X_FLAG': 'False'}):
254            self.assertFalse(env_bool('X_FLAG', default=True))
255        with mock.patch.dict(os.environ, {'X_FLAG': ' TRUE '}):
256            self.assertTrue(env_bool('X_FLAG'))
257        with mock.patch.dict(os.environ, {}, clear=True):
258            self.assertTrue(env_bool('X_FLAG', default=True))
259
260    def test_env_list(self):
261        """Separa por comas e ignora espacios y elementos vacíos."""
262        with mock.patch.dict(os.environ, {'X_LIST': 'a.com, b.com,,'}):
263            self.assertEqual(env_list('X_LIST'), ['a.com', 'b.com'])
264        with mock.patch.dict(os.environ, {}, clear=True):
265            self.assertEqual(env_list('X_LIST'), [])

279class SmtpKeycloakTests(SimpleTestCase):
280    """Pruebas de la configuración del correo de Keycloak a partir del correo de Django."""
281
282    def test_traduce_smtp_de_django_a_keycloak(self):
283        """Los datos de MAILERS y DEFAULT_FROM_EMAIL se pasan al formato smtpServer del realm."""
284        smtp = smtp_realm_desde_settings(SMTP_GMAIL, 'Global Exchange <grupo@gmail.com>')
285        self.assertEqual(smtp, {
286            'host': 'smtp.gmail.com', 'port': '587', 'from': 'grupo@gmail.com',
287            'fromDisplayName': 'Global Exchange', 'auth': 'true', 'user': 'grupo@gmail.com',
288            'password': 'clave-app', 'starttls': 'true', 'ssl': 'false',
289        })
290
291    def test_sin_remitente_usa_la_cuenta_smtp(self):
292        """Si DEFAULT_FROM_EMAIL está vacío, el remitente es la cuenta SMTP."""
293        smtp = smtp_realm_desde_settings(SMTP_GMAIL, '')
294        self.assertEqual(smtp['from'], 'grupo@gmail.com')
295        self.assertEqual(smtp['fromDisplayName'], 'Global Exchange')
296
297    def test_correo_por_consola_no_configura_keycloak(self):
298        """Con el backend de consola (desarrollo) no se toca el SMTP de Keycloak."""
299        consola = {'default': {'BACKEND': 'django.core.mail.backends.console.EmailBackend'}}
300        self.assertIsNone(smtp_realm_desde_settings(consola, 'x@x.com'))
301
302    @mock.patch.object(keycloak_bootstrap.requests, 'put')
303    def test_ensure_smtp_actualiza_el_realm(self, put):
304        """ensure_smtp hace un PUT parcial del realm solo con smtpServer."""
305        put.return_value = _resp()
306        kc = KeycloakAdmin('http://keycloak:8080', 'realm', 'client', 'secret')
307        kc.headers = {'Authorization': 'Bearer t'}
308        kc.ensure_smtp({'host': 'smtp.gmail.com'})
309        self.assertEqual(put.call_args.args[0], 'http://keycloak:8080/admin/realms/realm')
310        self.assertEqual(put.call_args.kwargs['json'], {'smtpServer': {'host': 'smtp.gmail.com'}})
311
312    @override_settings(
313        KEYCLOAK_REDIRECT_URI='https://ge.duckdns.org/auth/callback/',
314        MAILERS=SMTP_GMAIL, DEFAULT_FROM_EMAIL='Global Exchange <grupo@gmail.com>',
315    )
316    @mock.patch.object(KeycloakAdmin, 'ensure_smtp')
317    @mock.patch.object(KeycloakAdmin, 'ensure_client_urls', return_value=False)
318    @mock.patch.object(KeycloakAdmin, 'authenticate')
319    def test_comando_configura_smtp_si_hay_correo_real(self, _auth, _urls, smtp):
320        """El comando keycloak_bootstrap aplica el SMTP cuando Django usa SMTP."""
321        out = StringIO()
322        call_command('keycloak_bootstrap', '--sin-usuarios', stdout=out)
323        smtp.assert_called_once()
324        self.assertEqual(smtp.call_args.args[0]['host'], 'smtp.gmail.com')
325        self.assertIn('Correo de Keycloak configurado', out.getvalue())
