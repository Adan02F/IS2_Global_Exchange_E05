 30def _fecha(anio, mes, dia, hora=12):
 31    """Datetime con zona horaria local para fijar el ``timestamp`` de las operaciones de prueba."""
 32    return timezone.make_aware(datetime(anio, mes, dia, hora, 0), timezone.get_current_timezone())

 35def crear_operacion(modelo, fecha=None, desde='PYG', hacia='USD', convertido='0', tasa='1', comision='0',
 36                    iva='0', total='0', estado='SUCCESS', ventanilla=False, cliente=None):
 37    """
 38    Crea una compra o venta de prueba y, si se indica, fija su fecha (``timestamp`` es auto_now_add).
 39
 40    Args:
 41        modelo: ``CurrencyPurchaseTransaction`` o ``CurrencySaleTransaction``.
 42        fecha (datetime | None): Fecha de la operación; None deja la fecha actual.
 43        desde (str): Moneda de origen.
 44        hacia (str): Moneda de destino.
 45        convertido (str): Monto convertido.
 46        tasa (str): Tasa aplicada.
 47        comision (str): Comisión guardada (en la moneda que corresponda al canal).
 48        iva (str): IVA de la comisión guardado.
 49        total (str): Volumen en PYG.
 50        estado (str): Estado de la transacción.
 51        ventanilla (bool): True si es una operación de ventanilla.
 52        cliente (Cliente | None): Cliente de la operación.
 53
 54    Returns:
 55        Model: La transacción creada.
 56    """
 57    tx = modelo.objects.create(
 58        cliente=cliente, from_currency=desde, to_currency=hacia, amount=Decimal('1'),
 59        converted_amount=Decimal(convertido), applied_rate=Decimal(tasa), standard_rate=Decimal(tasa),
 60        commission_amount=Decimal(comision), tax_amount=Decimal(iva), total_pyg=Decimal(total), status=estado,
 61        payment_method_type=reportes.PAGO_VENTANILLA if ventanilla else 'TRANSFERENCIA',
 62    )
 63    if fecha is not None:
 64        modelo.objects.filter(pk=tx.pk).update(timestamp=fecha)
 65        tx.refresh_from_db()
 66    return tx
