211        """sol_cancelacion solo registra el pedido; la factura se cancela cuando SIFEN aprueba el evento."""
212        servicio.obtener_token.return_value = "token"
213        servicio.sol_cancelacion.return_value = {"code": 0, "description": "OK", "results": []}
214        tx = self.compra(horas=3)
