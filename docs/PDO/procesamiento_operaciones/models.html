 21def _segmento_sifen_actual(texto):
 22    """
 23    Devuelve la parte del desglose que corresponde al último intento de facturación.
 24
 25    El desglose es un registro que solo crece: cada reintento agrega la marca
 26    "SIFEN Reintento" y lo anterior queda como historial.
 27    """
 28    texto = texto or ""
 29    if MARCA_SIFEN_REINTENTO in texto:
 30        return texto.rsplit(MARCA_SIFEN_REINTENTO, 1)[1]
 31    return texto

 34def _ultimo_valor(texto, marca):
 35    """Valor que sigue a la última aparición de la marca, hasta el próximo separador '|'."""
 36    if marca not in texto:
 37        return None
 38    return texto.rsplit(marca, 1)[1].split("|")[0].strip()

 41class SifenFacturaMixin:
 42    """
 43    Propiedades de la factura electrónica (SIFEN) de una operación de compra o venta.
 44    El estado vigente es el último registrado en el intento de facturación actual.
 45    """
 46
 47    @property
 48    def sifen_cdc(self):
 49        """CDC de la factura del intento actual, o None si todavía no se emitió."""
 50        return _ultimo_valor(_segmento_sifen_actual(self.transparent_breakdown), MARCA_SIFEN_CDC) or None
 51
 52    @property
 53    def sifen_tiene_registro(self):
 54        """Indica si la operación tiene algún registro de facturación."""
 55        return MARCA_SIFEN_ESTADO in (self.transparent_breakdown or "")
 56
 57    @property
 58    def sifen_estado(self):
 59        """Estado vigente de la factura (ver ESTADOS_FACTURA). Sin registros se informa "Emitida"."""
 60        valor = _ultimo_valor(_segmento_sifen_actual(self.transparent_breakdown), MARCA_SIFEN_ESTADO)
 61        if not valor:
 62            return "Emitida"
 63        for estado in sorted(ESTADOS_FACTURA, key=len, reverse=True):
 64            if valor.startswith(estado):
 65                return estado
 66        return "Emitida"
 67
 68    @property
 69    def sifen_error(self):
 70        """Detalle del error de facturación vigente, o una cadena vacía si no hay error."""
 71        if self.sifen_estado != "Error":
 72            return ""
 73        valor = _ultimo_valor(_segmento_sifen_actual(self.transparent_breakdown), MARCA_SIFEN_ESTADO)
 74        return valor[len("Error"):].strip(" :")
 75
 76    @property
 77    def sifen_sin_confirmar(self):
 78        """
 79        True si el último envío a Factura Segura se cortó (timeout o red) y no se sabe si el
 80        documento se generó. Según la API ESI, hay que verificarlo antes de reenviarlo.
 81        """
 82        return self.sifen_estado == "Error" and self.sifen_error.startswith("Sin confirmar")
 83
 84    @property
 85    def sifen_detalle(self):
 86        """Texto completo del último estado registrado (incluye el motivo de rechazo o anulación)."""
 87        return _ultimo_valor(_segmento_sifen_actual(self.transparent_breakdown), MARCA_SIFEN_ESTADO) or ""
 88
 89    @property
 90    def tipo_operacion(self):
 91        """'COMPRA' o 'VENTA'."""
 92        return "VENTA" if self.__class__.__name__ == "CurrencySaleTransaction" else "COMPRA"
